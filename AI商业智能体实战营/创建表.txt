DROP TABLE station_device_coordinates;

-- 创建更新后的站场图基础设备坐标信息表
CREATE TABLE IF NOT EXISTS `station_device_coordinates` (
  `id` int(11) NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  `station_name` varchar(50) NOT NULL COMMENT '所属站点',
  `device_type` varchar(20) NOT NULL COMMENT '设备类型',
  `device_code` varchar(20) NOT NULL COMMENT '设备编号',
  `original_resolution` varchar(20) NOT NULL COMMENT '原始分辨率',
  `unit` varchar(10) NOT NULL COMMENT '坐标单位',
  `x_coordinate` int(11) NOT NULL COMMENT 'X轴坐标',
  `y_coordinate` int(11) NOT NULL COMMENT 'Y轴坐标',
  `distance` DECIMAL(10,2) DEFAULT NULL COMMENT '设备距离(米)',
  `deleted` bit(1) NOT NULL DEFAULT b'0' COMMENT '是否删除',
  `tenant_id` bigint NOT NULL DEFAULT '0' COMMENT '租户编号',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_station_device` (`station_name`, `device_code`) COMMENT '站点和设备编号唯一约束',
  KEY `idx_station` (`station_name`),
  KEY `idx_device_type` (`device_type`),
  KEY `idx_device_code` (`device_code`),
  KEY `idx_tenant` (`tenant_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='站场图设备坐标信息表';
